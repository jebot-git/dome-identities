cmake_minimum_required(VERSION 3.20)
project(dome-identities VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DOME_IDENTITIES_ENABLE_X11 "Enable explicit X11 appmenu integration" OFF)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(OpenSSL 3.0 REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig zlib)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
add_library(identity-core STATIC native/identities.cpp)
target_include_directories(identity-core PUBLIC native)
target_link_libraries(identity-core PUBLIC OpenSSL::Crypto PkgConfig::GIO Threads::Threads)
target_compile_options(identity-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dome-identities native/main.cpp)
target_link_libraries(dome-identities PRIVATE identity-core fltk::fltk-shared fltk::gl-shared fltk::images-shared OpenGL::GL PkgConfig::FONTCONFIG)
target_compile_options(dome-identities PRIVATE -Wall -Wextra -Wpedantic)
set(XNEDIT_ENABLE_X11 ${DOME_IDENTITIES_ENABLE_X11})
target_compile_definitions(dome-identities PRIVATE XNEDIT_ENABLE_X11=$<BOOL:${DOME_IDENTITIES_ENABLE_X11}>)
include(native/desktop-menu.cmake)
de_xr_desktop_menu(dome-identities)
set_target_properties(dome-identities PROPERTIES INSTALL_RPATH "/opt/fltk-vr/toolkit/lib")
install(TARGETS dome-identities RUNTIME DESTINATION bin)
install(FILES packaging/org.domed.Identities.desktop DESTINATION share/applications)
install(FILES packaging/org.domed.Identities.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES README.md LICENSE DESTINATION share/doc/dome-identities)
include(CTest)
if(BUILD_TESTING)
 add_executable(identity-tests tests/core.cpp)
 target_link_libraries(identity-tests PRIVATE identity-core)
 target_compile_options(identity-tests PRIVATE -UNDEBUG)
 add_test(NAME isolated-identity-management COMMAND identity-tests)
 add_executable(penguin-identity-tests tests/penguin.cpp)
 target_link_libraries(penguin-identity-tests PRIVATE identity-core)
 target_compile_options(penguin-identity-tests PRIVATE -UNDEBUG)
 add_test(NAME isolated-penguin-identity-import COMMAND penguin-identity-tests)
 add_executable(store-choice-tests tests/store-choice.cpp)
 target_include_directories(store-choice-tests PRIVATE native)
 target_link_libraries(store-choice-tests PRIVATE fltk::fltk-shared)
 target_compile_options(store-choice-tests PRIVATE -UNDEBUG)
 add_test(NAME flat-selectable-identity-stores COMMAND store-choice-tests)
endif()
